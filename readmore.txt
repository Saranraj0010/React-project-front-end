import { useState } from "react";

interface ReadMoreParagraphProps {
  text: string;
  limit?: number;
}

const ReadMoreParagraph: React.FC<ReadMoreParagraphProps> = ({ text, limit = 100 }) => {
  const [isExpanded, setIsExpanded] = useState(false);
  const itCanOverflow = text.length > limit;
  const displayedText = isExpanded || !itCanOverflow? text: `${text.substring(0, limit)}...`;

  const toggleReadMore = () => {
    setIsExpanded(!isExpanded);
  };

  return (
    <div className=" font-semibold text-sm text-black/80 px-3 ">
      <p>
        {displayedText}
      </p>
      {itCanOverflow && (
        <button
          onClick={toggleReadMore}
          className="text-sm text-gray-500 underline cursor-pointer decoration-0"
        >
          {isExpanded ? 'Read Less' : 'Read More'}
        </button>
      )}
    </div>
  );
};

export default ReadMoreParagraph;
